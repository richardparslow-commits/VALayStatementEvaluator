# Development-only dependencies (not needed to run the app).
# Install with:  pip install -r requirements-dev.txt
# CI installs this after requirements.lock (see .github/workflows/test.yml).
-r requirements.txt
mypy>=1.18
types-requests
# Optional tracing backend, installed here so the tracing tests exercise a real
# TracerProvider and in-memory span exporter instead of skipping (the app itself
# no-ops without these — see requirements-otel.txt).
opentelemetry-sdk>=1.44
opentelemetry-exporter-otlp-proto-http>=1.44
# S3 client for the backup tests only. tests/test_audit_backup_s3.py drives the
# real boto3 backend against a loopback S3 server, so installing this makes those
# assertions run rather than skip — a skipped backup test would mean "unverified"
# for the code path that has to survive a pod failure.
# The GCS and Azure backends are optional too; they are covered by interface tests
# only (see tests/test_audit_backup.py), not against a live account.
boto3>=1.34
# YAML parser for the deployment-asset tests. tests/test_monitoring_assets.py
# parses the Prometheus rules / Grafana dashboards / k8s manifests with it and
# skips itself without it — which in CI meant 24 assertions that never ran, so
# a broken scrape config or dashboard JSON could reach main unverified.
pyyaml>=6.0

# NOTE: the Perplexity SDK is not listed here. It used to be a dev-only addition on top
# of the optional install, but it now arrives with `-r requirements.txt` because the
# Research tab has to work on hosts that cannot install at runtime (Streamlit Community
# Cloud). tests/test_perplexity_agent.py therefore drives the real SDK objects everywhere,
# including CI, instead of skipping.
